    <script>
        // Función para copiar correo al portapapeles
        function copyToClipboard(text) {
            const tempInput = document.createElement("input");
            tempInput.value = text;
            document.body.appendChild(tempInput);
            tempInput.select();
            document.execCommand("copy");
            document.body.removeChild(tempInput);

            const notif = document.getElementById('form-notification');
            notif.className = 'mb-6 p-4 rounded-xl text-center text-sm font-medium bg-brand-500/20 text-brand-300 border border-brand-500/30';
            notif.textContent = `¡Correo "${text}" copiado al portapapeles!`;
            notif.classList.remove('hidden');

            setTimeout(() => {
                notif.classList.add('hidden');
            }, 3500);
        }

        document.addEventListener('DOMContentLoaded', () => {

            // --- Menú Móvil ---
            const mobileMenuBtn = document.getElementById('mobile-menu-btn');
            const mobileMenu = document.getElementById('mobile-menu');

            mobileMenuBtn.addEventListener('click', () => {
                mobileMenu.classList.toggle('hidden');
            });

            // --- Lógica del Cotizador Interactivo ---
            const projectTypeCards = document.querySelectorAll('.project-type-card');
            const addonCheckboxes = document.querySelectorAll('.addon-checkbox input[type="checkbox"]');
            
            const summaryType = document.getElementById('summary-type');
            const summaryTime = document.getElementById('summary-time');
            const totalPriceEl = document.getElementById('total-price');
            const selectedServiceInput = document.getElementById('selected-service-input');

            let basePrice = 250;
            let currentTypeName = 'Landing Page';
            let currentTime = '3 a 5 días';

            function updateQuoter() {
                let addonsTotal = 0;

                addonCheckboxes.forEach(cb => {
                    if (cb.checked) {
                        addonsTotal += parseFloat(cb.dataset.price);
                    }
                });

                const grandTotal = basePrice + addonsTotal;

                summaryType.textContent = currentTypeName;
                summaryTime.textContent = currentTime;
                totalPriceEl.textContent = `$${grandTotal}`;
                selectedServiceInput.value = `${currentTypeName} (PixelForge) + Extras - Total $${grandTotal} USD`;
            }

            // Selección de Tipo de Proyecto
            projectTypeCards.forEach(card => {
                card.addEventListener('click', () => {
                    projectTypeCards.forEach(c => {
                        c.classList.remove('border-brand-500', 'bg-brand-600/10');
                        c.classList.add('border-darkborder', 'bg-darkbg/50');
                    });

                    card.classList.remove('border-darkborder', 'bg-darkbg/50');
                    card.classList.add('border-brand-500', 'bg-brand-600/10');

                    const radio = card.querySelector('input[type="radio"]');
                    radio.checked = true;

                    basePrice = parseFloat(card.dataset.price);
                    currentTime = card.dataset.time;

                    const typeValue = card.dataset.type;
                    if (typeValue === 'landing') currentTypeName = 'Landing Page';
                    else if (typeValue === 'corporate') currentTypeName = 'Sitio Corporativo';
                    else if (typeValue === 'ecommerce') currentTypeName = 'E-Commerce / Tienda';
                    else if (typeValue === 'custom') currentTypeName = 'Desarrollo a Medida';

                    updateQuoter();
                });
            });

            if (projectTypeCards.length > 0) {
                projectTypeCards[0].classList.add('border-brand-500', 'bg-brand-600/10');
            }

            addonCheckboxes.forEach(cb => {
                cb.addEventListener('change', updateQuoter);
            });

            // --- Envío de Formulario Simulado ---
            const contactForm = document.getElementById('contact-form');
            const notification = document.getElementById('form-notification');

            contactForm.addEventListener('submit', (e) => {
                e.preventDefault();

                const name = document.getElementById('form-name').value;

                notification.className = 'mb-6 p-4 rounded-xl text-center text-sm font-medium bg-emerald-500/20 text-accent-emerald border border-accent-emerald/30';
                notification.innerHTML = `¡Gracias ${name}! Hemos recibido tu cotización en <strong>PixelForge Web</strong>. Te responderemos a la brevedad desde <span class="underline">pixelforge.web08@gmail.com</span>.`;
                notification.classList.remove('hidden');

                contactForm.reset();
                updateQuoter();

                setTimeout(() => {
                    notification.classList.add('hidden');
                }, 7000);
            });

            // Botón WhatsApp Directo
            document.getElementById('btn-whatsapp-direct').addEventListener('click', () => {
                const name = document.getElementById('form-name').value || 'Hola';
                const service = selectedServiceInput.value;
                const message = encodeURIComponent(`${name}, me interesa cotizar con PixelForge Web el servicio: ${service}`);
                window.open(`https://wa.me/?text=${message}`, '_blank');
            });

            // Función global para seleccionar plan
            window.selectPlan = function(planName) {
                selectedServiceInput.value = planName;
            };

        });
    </script>
